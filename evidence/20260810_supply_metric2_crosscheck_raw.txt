## 一時スクリプトの内容（実行後に削除済み。tests/test_gtfs_supply_metrics.py には含まれない）

"""SUPPLY-METRIC-2の再検証用: src/calculate_gtfs_supply_metrics.py とは独立に書いた
手組みの計算で、2026-04-06..2026-04-12のscheduled_trip_count_by_dateを再現する。
一時スクリプト。実行後にリポジトリから削除する。読み取り専用、ネットワーク接続なし。
"""
import csv
import io
import zipfile
from collections import Counter
from datetime import date, timedelta

WEEKDAY_COLS = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"]
DATES = [date(2026, 4, 6) + timedelta(days=i) for i in range(7)]


def read_csv(zf, name):
    raw = zf.read(name)
    if raw.startswith(b"\xef\xbb\xbf"):
        raw = raw[3:]
    text = raw.decode("utf-8")
    rows = list(csv.reader(io.StringIO(text)))
    return rows[0], rows[1:]


for label, path in [
    ("iwakuni-gtfsjp", "raw/gtfs/iwakuni_gtfsjp_20260401.zip"),
    ("hikari-gtfs", "raw/gtfs/hikari_gtfs_20260401.zip"),
]:
    print("=" * 20, label, "=" * 20)
    with zipfile.ZipFile(path) as zf:
        cal_header, cal_rows = read_csv(zf, "calendar.txt")
        cd_header, cd_rows = read_csv(zf, "calendar_dates.txt")
        trips_header, trips_rows = read_csv(zf, "trips.txt")

        sid_i = trips_header.index("service_id")
        trip_counts = Counter(r[sid_i] for r in trips_rows)

        cal_sid_i = cal_header.index("service_id")
        start_i = cal_header.index("start_date")
        end_i = cal_header.index("end_date")
        weekday_idx = {d: cal_header.index(d) for d in WEEKDAY_COLS}

        cd_sid_i = cd_header.index("service_id")
        cd_date_i = cd_header.index("date")
        cd_exc_i = cd_header.index("exception_type")

        for d in DATES:
            dstr = d.strftime("%Y%m%d")
            weekday_name = WEEKDAY_COLS[d.weekday()]
            valid = set()
            for row in cal_rows:
                sid = row[cal_sid_i]
                start = row[start_i]
                end = row[end_i]
                flag = row[weekday_idx[weekday_name]]
                if start <= dstr <= end and flag == "1":
                    valid.add(sid)
            for row in cd_rows:
                if row[cd_date_i] != dstr:
                    continue
                sid = row[cd_sid_i]
                exc = row[cd_exc_i]
                if exc == "1":
                    valid.add(sid)
                elif exc == "2":
                    valid.discard(sid)
            total = sum(trip_counts.get(sid, 0) for sid in valid)
            print(f"{d.isoformat()} ({weekday_name}): trip_total={total}")

        print(f"agency.txt rows: {len(read_csv(zf, 'agency.txt')[1])}")
        r_header, r_rows = read_csv(zf, "routes.txt")
        r_id_i = r_header.index("route_id")
        r_ids = [r[r_id_i] for r in r_rows]
        print(f"routes.txt unique route_id: {len(set(r_ids))} (rows={len(r_rows)}, blank={sum(1 for v in r_ids if v.strip()=='')})")
        s_header, s_rows = read_csv(zf, "stops.txt")
        s_id_i = s_header.index("stop_id")
        s_lt_i = s_header.index("location_type") if "location_type" in s_header else None
        if s_lt_i is not None:
            boarding = sum(1 for row in s_rows if row[s_lt_i].strip() in ("", "0"))
        else:
            boarding = len(s_rows)
        print(f"stops.txt boarding-location stop_id count (location_type blank/0): {boarding} (rows={len(s_rows)})")
        print(f"frequencies.txt present: {'frequencies.txt' in zf.namelist()}")
